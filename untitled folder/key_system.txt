local function _crc(d)

        c = bit32.bxor(bit32.rshift(c, 8), _crc_t[bit32.band(bit32.bxor(c, b), 0xFF)])
    end
    return bit32.bxor(c, 0xFFFFFFFF)
end

local function _derive(s, n)
    for _ = 1, n or 32 do
        s = bit32.bxor(s, bit32.lshift(s, 13))
        s = bit32.bxor(s, bit32.rshift(s, 17))
        s = bit32.bxor(s, bit32.lshift(s, 5))
    end
    return string.format("%08X", s)
end

        p[i] = _derive(m, 16)
    end

end
