local _DELTA = 0x9E3779B9
local _ROUNDS = 32

local function _xtea_enc(v0, v1, k)
    local s = 0
    for _ = 1, _ROUNDS do
        v0 = v0 + bit32.bxor(bit32.bxor(bit32.lshift(v1,4),bit32.rshift(v1,5))+v1, s+k[bit32.band(s,3)+1])
        v0 = bit32.band(v0, 0xFFFFFFFF)
        s = bit32.band(s + _DELTA, 0xFFFFFFFF)
        v1 = v1 + bit32.bxor(bit32.bxor(bit32.lshift(v0,4),bit32.rshift(v0,5))+v0, s+k[bit32.band(bit32.rshift(s,11),3)+1])
        v1 = bit32.band(v1, 0xFFFFFFFF)
    end
    return v0, v1
end

local function _xtea_dec(v0, v1, k)
    local s = bit32.band(_DELTA * _ROUNDS, 0xFFFFFFFF)
    for _ = 1, _ROUNDS do
        v1 = v1 - bit32.bxor(bit32.bxor(bit32.lshift(v0,4),bit32.rshift(v0,5))+v0, s+k[bit32.band(bit32.rshift(s,11),3)+1])
        v1 = bit32.band(v1, 0xFFFFFFFF)
        s = bit32.band(s - _DELTA, 0xFFFFFFFF)
        v0 = v0 - bit32.bxor(bit32.bxor(bit32.lshift(v1,4),bit32.rshift(v1,5))+v1, s+k[bit32.band(s,3)+1])
        v0 = bit32.band(v0, 0xFFFFFFFF)
    end
    return v0, v1
end

local _k = {0x01234567, 0x89ABCDEF, 0xFEDCBA98, 0x76543210}
local _e0, _e1 = _xtea_enc(0xDEADBEEF, 0xCAFEBABE, _k)
local _d0, _d1 = _xtea_dec(_e0, _e1, _k)

local _POLY = 0xEDB88320
local _ct = {}
for i = 0, 255 do
    local c = i
    for _ = 1, 8 do
        if bit32.band(c,1)==1 then c = bit32.bxor(bit32.rshift(c,1),_POLY)
        else c = bit32.rshift(c,1) end
    end
    _ct[i] = c
end

local function _crc(d)
    local c = 0xFFFFFFFF
    for i = 1, #d do
        c = bit32.bxor(bit32.rshift(c,8), _ct[bit32.band(bit32.bxor(c, string.byte(d,i)), 0xFF)])
    end
    return bit32.bxor(c, 0xFFFFFFFF)
end

local function _gen_wm()
    local p = {}; local cs = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"
    local st = os.time()
    for i = 1, 4 do
        local seg = ""
        for _ = 1, 8 do
            st = bit32.bxor(st, bit32.lshift(st,13))
            st = bit32.bxor(st, bit32.rshift(st,17))
            st = bit32.bxor(st, bit32.lshift(st,5))
            seg = seg .. cs:sub((st%#cs)+1, (st%#cs)+1)
        end
        p[i] = seg
    end
    return table.concat(p, "-")
end

local function _fnv1a(d)
    local h = 0x811C9DC5
    for i = 1, #d do
        h = bit32.bxor(h, string.byte(d,i))
        h = bit32.band(h * 0x01000193, 0xFFFFFFFF)
    end
    return h
end

local function _sign(msg, key)
    local inner = _fnv1a(key .. msg)
    local outer = _fnv1a(key .. string.format("%08X", inner))
    return string.format("%08X%08X", inner, outer)
end
