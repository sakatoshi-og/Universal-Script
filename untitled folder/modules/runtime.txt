local _hook_sigs = {
    "hookfunction","hookmetamethod","newcclosure","getnamecallmethod",
    "checkcaller","getrawmetatable","setreadonly","iscclosure",

local function _detect()
    local score = 0
    local env = getfenv and getfenv(0) or {}
    for _, sig in ipairs(_hook_sigs) do
        if rawget(env, sig) ~= nil then score = score + 1 end
    end
    return score
end

local _vmr = {}; for i = 0, 15 do _vmr[i] = 0 end
local _vmo = {
    [0x00] = function(a,b) _vmr[a] = b end,
    [0x01] = function(a,b) _vmr[a] = _vmr[a] + _vmr[b] end,
    [0x02] = function(a,b) _vmr[a] = bit32.bxor(_vmr[a], _vmr[b]) end,
    [0x03] = function(a,_) _vmr[a] = bit32.bnot(_vmr[a]) end,
    [0x04] = function(a,b) _vmr[a] = bit32.lshift(_vmr[a], b) end,
    [0x05] = function(a,b) _vmr[a] = bit32.rshift(_vmr[a], b) end,
    [0x06] = function(a,b) _vmr[a] = bit32.band(_vmr[a], _vmr[b]) end,
    [0x07] = function(a,b) _vmr[a] = bit32.bor(_vmr[a], _vmr[b]) end,
}

local function _vmexec()
    for _, i in ipairs(_prog) do
        local op = _vmo[i[1]]
        if op then op(i[2], i[3]) end
    end
    return _vmr[0]
end

local _bt = {v=50,l=nil,r=nil}
local function _bti(n,val)
    if not n then return {v=val,l=nil,r=nil} end
    if val < n.v then n.l = _bti(n.l, val)
    elseif val > n.v then n.r = _bti(n.r, val) end
    return n
end
local function _btt(n,res)
    if not n then return end
    res = res or {}
    _btt(n.l, res); table.insert(res, n.v); _btt(n.r, res)
    return res
end
for _,v in ipairs({25,75,12,37,62,87,6,18,31,43,56,68,81,93}) do
    _bt = _bti(_bt, v)
end

local _st = {}
local function _timing()
    local s = tick(); local d = 0
    for i = 1, 500 do d = bit32.bxor(d, i) end
    local e = tick() - s
    table.insert(_st, e)
    if #_st > 32 then table.remove(_st, 1) end
    if #_st >= 8 then
        local avg = 0
        for _, v in ipairs(_st) do avg = avg + v end
        avg = avg / #_st
        if e > avg * 3 then return false end
    end
    return true
end
